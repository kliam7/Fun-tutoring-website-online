            'tetris': () => {
                const div = document.createElement('div');
                div.innerHTML = '<h3>Block Puzzle (Local)</h3><canvas width="200" height="400"></canvas><p>Score: <span id="score">0</span> Use arrows to move, up to rotate!</p>';
                const canvas = div.querySelector('canvas');
                const ctx = canvas.getContext('2d');
                let grid = Array(20).fill().map(() => Array(10).fill(0));
                let piece = null, score = 0;
                const shapes = [
                    [[1,1,1,1]], // I
                    [[1,1],[1,1]], // O
                    [[0,1,0],[1,1,1]], // T
                    [[1,1,0],[0,1,1]], // S
                    [[0,1,1],[1,1,0]] // Z
                ];
                function spawnPiece() {
                    const shape = shapes[Math.floor(Math.random() * shapes.length)];
                    piece = {x:4,y:0,shape:shape.map(row => [...row])};
                }
                function collide() {
                    for (let y = 0; y < piece.shape.length; y++) for (let x = 0; x < piece.shape[y].length; x++) {
                        if (piece.shape[y][x] && (grid[piece.y + y]?.[piece.x + x] || piece.y + y >= 20 || piece.x + x < 0 || piece.x + x >= 10)) return true;
                    }
                    return false;
                }
                function merge() {
                    for (let y = 0; y < piece.shape.length; y++) for (let x = 0; x < piece.shape[y].length; x++) {
                        if (piece.shape[y][x]) grid[piece.y + y][piece.x + x] = 1;
                    }
                }
                function clearLines() {
                    grid = grid.filter(row => row.some(cell => !cell));
                    while (grid.length < 20) grid.unshift(Array(10).fill(0));
                    score += 100;
                }
                function loop() {
                    ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--card-bg');
                    ctx.fillRect(0,0,200,400);
                    if (!piece) spawnPiece();
                    piece.y++;
                    if (collide()) {
                        piece.y--;
                        merge();
                        clearLines();
                        spawnPiece();
                        if (collide()) {
                            grid = Array(20).fill().map(() => Array(10).fill(0));
                            score = 0;
                            spawnPiece();
                        }
                    }
                    for (let y = 0; y < 20; y++) for (let x = 0; x < 10; x++) {
                        ctx.fillStyle = grid[y][x] ? getComputedStyle(document.documentElement).getPropertyValue('--accent-color') : '#eee';
                        ctx.fillRect(x*20, y*20, 18, 18);
                    }
                    if (piece) {
                        ctx.fillStyle = getComputedStyle(document.documentElement).getPropertyValue('--accent-color');
                        for (let y = 0; y < piece.shape.length; y++) for (let x = 0; x < piece.shape[y].length; x++) {
                            if (piece.shape[y][x]) ctx.fillRect((piece.x + x)*20, (piece.y + y)*20, 18, 18);
                        }
                    }
                    div.querySelector('#score').textContent = score;
                    div.querySelector('#score').style.color = getComputedStyle(document.documentElement).getPropertyValue('--text-color');
                    setTimeout(loop, 500);
                }
                loop();
                document.addEventListener('keydown', (e) => {
                    if (!piece) return;
                    let oldX = piece.x, oldY = piece.y, oldShape = piece.shape.map(row => [...row]);
                    if (e.key === 'ArrowLeft') piece.x--;
                    else if (e.key === 'ArrowRight') piece.x++;
                    else if (e.key === 'ArrowDown') piece.y++;
                    else if (e.key === 'ArrowUp') {
                        piece.shape = piece.shape[0].map((_,i) => piece.shape.map(row => row[row.length-1-i]));
                    }
                    if (collide()) {
                        piece.x = oldX; piece.y = oldY; piece.shape = oldShape;
                    }
                });
                return div;
            },

                const savedSpeed = localStorage.getItem('animation-speed') || '0.3s';
                document.getElementById('animation-speed').value = savedSpeed;
                updateAnimationSpeed();
                particlesEnabled = (localStorage.getItem('particles-enabled') || 'true') === 'true';
                particleDensity = parseInt(localStorage.getItem('particle-density') || '30');
                document.getElementById('particles-enabled').value = particlesEnabled ? 'true' : 'false';
                document.getElementById('particle-density').value = particleDensity;
                updateParticles();
            } catch (e) {
                console.log('LocalStorage blocked—using defaults');
            }
        };

        function getProxiedUrl(toolUrl, proxyIndex) {
            return proxies[proxyIndex].base + encodeURIComponent(toolUrl);
        }

        function testProxy(index) {
            const select = document.getElementById(`proxy-select-${index}`);
            const proxyIdx = parseInt(select.value);
            const status = document.getElementById(`status-${index}`);
            status.textContent = 'Testing...';
            status.style.color = getComputedStyle(document.documentElement).getPropertyValue('--text-color');
            const testIframe = document.createElement('iframe');
            testIframe.style.display = 'none';
            testIframe.src = getProxiedUrl(tools[index].url, proxyIdx);
            document.body.appendChild(testIframe);
            const timeout = setTimeout(() => {
                status.textContent = `${proxies[proxyIdx].name}: Timed out`;
                status.style.color = '#e53935';
                document.body.removeChild(testIframe);
            }, 1500);
            testIframe.onload = () => {
                clearTimeout(timeout);
                try {
                    const content = testIframe.contentWindow.document;
                    if (content.title.includes('404') || content.title.includes('Error') || content.body.innerHTML.includes('GoGuardian') || content.body.innerHTML.length < 200) {
                        status.textContent = `${proxies[proxyIdx].name}: Failed`;
                        status.style.color = '#e53935';
                    } else {
                        status.textContent = `${proxies[proxyIdx].name}: Working!`;
                        status.style.color = '#43a047';
                    }
                } catch (e) {
                    status.textContent = `${proxies[proxyIdx].name}: Blocked`;
                    status.style.color = '#e53935';
                }
                document.body.removeChild(testIframe);
            };
        }

        function playTool(index, proxyIdx = 0) {
            currentToolIndex = index;
            currentProxyIndex = proxyIdx;
            toolList.style.display = 'none';
            loading.style.display = 'block';
            error.style.display = 'none';
            toolPlayer.innerHTML = '';
            musicPlayer.classList.remove('active');
            musicPlayer.innerHTML = '';
            videoPlayer.classList.remove('active');
            videoPlayer.innerHTML = '';
            if (tools[index].local === 'music') {
                toolPlayer.style.display = 'none';
                videoPlayer.style.display = 'none';
                musicPlayer.appendChild(localGames['music']());
                musicPlayer.classList.add('active');
                backBtn.style.display = 'block';
                loading.style.display = 'none';
            } else if (tools[index].local === 'video') {
                toolPlayer.style.display = 'none';
                musicPlayer.style.display = 'none';
                videoPlayer.appendChild(localGames['video']());
                videoPlayer.classList.add('active');
                backBtn.style.display = 'block';
                loading.style.display = 'none';
            } else {
                toolPlayer.src = getProxiedUrl(tools[index].url, proxyIdx);
                toolPlayer.style.display = 'block';
                backBtn.style.display = 'block';

                const checkIframe = () => {
                    try {
                        const content = toolPlayer.contentWindow.document;
                        if (!content || content.title.includes('404') || content.title.includes('Error') || content.body.innerHTML.includes('GoGuardian') || content.body.innerHTML.includes('blocked') || content.body.innerHTML.length < 200) {
                            throw new Error('Failed load');
                        }
                        loading.style.display = 'none';
                    } catch (e) {
                        tryNextProxy();
                    }
                };

                toolPlayer.onload = checkIframe;
                setTimeout(checkIframe, 1500);
            }
        }

        function tryNextProxy() {
            if (currentProxyIndex < proxies.length - 1) {
                currentProxyIndex++;
                setTimeout(() => {
                    error.style.display = 'block';
                    loading.style.display = 'block';
                    toolPlayer.src = getProxiedUrl(tools[currentToolIndex].url, currentProxyIndex);
                    toolPlayer.onload = () => setTimeout(checkIframeFromTry, 1500);
                }, 500 + Math.random() * 500);
            } else {
                loading.style.display = 'none';
                error.style.display = 'block';
                toolPlayer.innerHTML = '';
                const localKey = tools[currentToolIndex].local;
                if (localGames[localKey]) {
                    toolPlayer.appendChild(localGames[localKey]());
                    error.style.display = 'none';
                } else {
                    toolPlayer.innerHTML = '<p>Local version not available—try fallback link.</p>';
                }
            }
        }

        function checkIframeFromTry() {
            try {
                const content = toolPlayer.contentWindow.document;
                if (!content || content.title.includes('404') || content.title.includes('Error') || content.body.innerHTML.includes('GoGuardian') || content.body.innerHTML.length < 200) {
                    tryNextProxy();
                } else {
                    loading.style.display = 'none';
                    error.style.display = 'none';
                }
            } catch (e) {
                tryNextProxy();
            }
        }

        function skipToLocal() {
            loading.style.display = 'none';
            error.style.display = 'none';
            toolPlayer.innerHTML = '';
            const localKey = tools[currentToolIndex].local;
            if (localGames[localKey]) {
                toolPlayer.appendChild(localGames[localKey]());
            }
        }

        function backToList() {
            toolPlayer.src = '';
            toolPlayer.innerHTML = '';
            toolPlayer.style.display = 'none';
            musicPlayer.classList.remove('active');
            musicPlayer.innerHTML = '';
            videoPlayer.classList.remove('active');
            videoPlayer.innerHTML = '';
            toolList.style.display = 'grid';
            backBtn.style.display = 'none';
            loading.style.display = 'none';
            error.style.display = 'none';
        }
